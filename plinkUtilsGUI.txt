import psutil


def measure_operation(operation, form):
    """Measure elapsed time and sampled process-tree RSS; append a CSV record.

    A normal return does not establish analysis success: backend functions may
    print an error and return without raising. Check the analysis output/log.
    """
    parent = psutil.Process()
    stop = threading.Event()
    interval = 0.1
    peak_bytes = 0
    samples = 0
    sample_errors = 0

    def sample():
        nonlocal peak_bytes, samples, sample_errors
        try:
            processes = [parent] + parent.children(recursive=True)
        except psutil.Error:
            processes = [parent]
            sample_errors += 1
        total = 0
        for process in processes:
            try:
                total += process.memory_info().rss
            except psutil.NoSuchProcess:
                pass
            except psutil.Error:
                sample_errors += 1
        peak_bytes = max(peak_bytes, total)
        samples += 1

    def monitor():
        while not stop.wait(interval):
            sample()

    # Capture selections before the operation; backend code may change cwd.
    output_dir = os.path.abspath(form.w_lbl_outdir.text())
    metadata = {
        "started_at": datetime.now().astimezone().isoformat(timespec="seconds"),
        "operation": operation.__name__,
        "root_name": form.w_root_name.text() or "plink",
        "chromosome": form.w_combo01.currentText(),
        "bp_lower": form.w_bp_lower.text(),
        "bp_upper": form.w_bp_upper.text(),
        "output_directory": output_dir,
        "python_version": sys.version.split()[0],
        "psutil_version": psutil.__version__,
    }
    sample()
    baseline_bytes = peak_bytes
    worker = threading.Thread(target=monitor, daemon=True)
    worker.start()
    status = "returned; success requires output verification"
    start = time.perf_counter()
    try:
        return operation(form)
    except BaseException as exc:
        status = "raised " + type(exc).__name__
        raise
    finally:
        elapsed = time.perf_counter() - start
        stop.set()
        worker.join()
        sample()
        row = dict(metadata,
                   elapsed_seconds=round(elapsed, 6),
                   sampled_peak_rss_mib=round(peak_bytes / 1024**2, 3),
                   baseline_rss_mib=round(baseline_bytes / 1024**2, 3),
                   requested_sampling_interval_seconds=interval,
                   sample_count=samples,
                   sampling_errors=sample_errors,
                   status=status)
        print("\nPERFORMANCE MEASUREMENT")
        print(f"Operation: {operation.__name__}")
        print(f"Elapsed time: {elapsed:.3f} seconds")
        print(f"Sampled peak resident memory: {peak_bytes / 1024**2:.1f} MiB")
        print(f"Baseline resident memory: {baseline_bytes / 1024**2:.1f} MiB")
        print(f"Requested sampling interval: {interval} seconds")
        print(f"Status: {status}")
        csv_path = os.path.join(output_dir, "performance_measurements.csv")
        try:
            needs_header = not os.path.exists(csv_path) or os.path.getsize(csv_path) == 0
            with open(csv_path, "a", newline="", encoding="utf-8") as handle:
                writer = csv.DictWriter(handle, fieldnames=list(row))
                if needs_header:
                    writer.writeheader()
                writer.writerow(row)
            print("Measurements saved to:", csv_path)
        except OSError as exc:
            print("Could not save measurements:", exc)

        hbox = QHBoxLayout()

    def fetchOutDir(self):
        fname_cur = self.w_lbl_outdir.text()
        fname = QFileDialog.getExistingDirectory(self, "Select directory", fname_cur)
        if fname != "" and fname != fname_cur:
            fname = os.path.normpath(fname)
            self.w_lbl_outdir.setText(fname)

    def markerExtraction(self):
        if validate_inputs(self):
            measure_operation(extract_markers, self)

    def runLdDecay(self):
        if validate_inputs(self):
            measure_operation(ld_decay, self)

    def runLdDecayGraph(self):
        if validate_inputs(self):
            measure_operation(ld_decay_graph, self)

    def runPotentialHaplotype(self):
        if validate_inputs(self):
            measure_operation(potential_haplotype, self)

    def runFHBD(self):
        if validate_inputs(self):
            measure_operation(formal_haplotype_block_detection, self)

    def runTwoSnpAnova(self):
        if validate_inputs(self):
            measure_operation(two_snp_pair_anova, self)

    def runSnpTraitAnalysis(self):
        if validate_inputs(self):
            measure_operation(snp_trait_analysis, self)

    def exitClicked(self):
        write_config_file(self)
        self.close()

    app = QApplication(sys.argv)
    form = Form()
    form.show()
    sys.exit(app.exec_())


if __name__ == "__main__":
    main()
